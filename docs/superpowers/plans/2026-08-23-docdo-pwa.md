# DocDo PWA Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 어르신이 우편물을 찍으면 Upstage Studio가 판별하고 우리 검증 계층이 대조해, 큰 글씨와 음성으로 알려주고 자녀 화면에 즉시 등록하는 웹앱을 만든다.

**Architecture:** Next.js 15 App Router 단일 앱. `/elder`와 `/guardian` 두 화면이 같은 배포 주소를 공유한다. API Route가 Upstage Agent API를 중계하되 **job 생성과 조회를 분리**해(요청당 Upstage 1회 조회) 서버가 완료를 기다리지 않는다. 검증 계층은 이미 Python으로 검증된 규칙(R1/R3/R4/R5/R6)을 TypeScript로 포팅하며, 공격·결손 입력 19종 회귀 시험을 그대로 옮긴다.

**Tech Stack:** Next.js 15 (App Router, TypeScript) · Vercel · Supabase (Postgres) · Vitest · Web Speech API (`speechSynthesis`) · Upstage Agent API v2

**Spec:** `docs/superpowers/specs/2026-08-22-docdo-design.md`

## Global Constraints

- **제출 마감 2026-08-23 12:00 KST.** 11:00에 제출을 시작한다.
- **`UPSTAGE_API_KEY`는 서버 환경변수에만.** `NEXT_PUBLIC_` 접두사 절대 금지 — 소스 전체 공개가 대회 규칙이다.
- **API Route 안에서 `while`/`sleep` 폴링 금지.** job 생성 후 즉시 반환, GET 요청당 Upstage 1회 조회.
- **`output[].model` 정규식으로 단계 탐색.** 배열 순번(`output[2]`) 접근 금지 — 분기로 단계가 생략된다.
- **`include: ["all"]`** 로 호출한다. `["last"]`면 Extract 필드·confidence·location이 사라진다.
- **`additional_values`는 객체로 온다** (문자열이면 `JSON.parse`).
- **Agent ID:** `agt_PNZaixnk4TZxkDg5muEJQi` · **config:** #4 (저장본. 초안은 API에서 안 보인다)
- **허용목록 밖 필드는 저장하지 않는다.** `account_number`·`recipient_name` 등은 폐기.
- **어르신 화면에 금전 거래 경로를 두지 않는다** — 계좌번호·납부 버튼·결제 링크 없음.
- **`low`·결손이면 숫자를 읽지 않는다. 지시하지 않고 사실만 전달한다.**
- **문서에서 추출한 연락처·링크는 실행 경로가 되지 못한다** — 레지스트리 값만 `tel:`.
- **판정 표기:** `clear`는 "확인된 불일치 없음". **"정상"·"진짜"·"안전" 표현 금지.**
- 사진은 1장, JPEG, 긴 변 ~1600px, 2~3MB 이하로 압축.

- [x] **Step 6: 검증 계층 구현**

`lib/verify.ts` — `scripts/verify.py`의 규칙을 그대로 옮긴다. 정규식의 제어문자 범위는 반드시 `\x00-\x1f` 이스케이프로 쓴다.

```ts
import { REGISTRY } from "./registry";
import type { Check, Issuer, VerifyResult, Verdict } from "./types";

const BRANCH_TAIL = /^([가-힣A-Za-z0-9]{0,12}(지사|지역본부|본부|지점|센터|출장소|사무소|과|팀|담당관|반|부))+$/;
const MIN_ALIAS = 3;
const MOBILE = /^01[016789]/;
const CTRL = /[\x00-\x1f\x7f]/;
const ALLOWED_SCHEME = new Set(["http", "https"]);
const ALLOW_FIELDS = new Set(["issuer", "doc_title", "epn", "amount_krw", "issue_date", "due_date",
  "contact_phone", "info_url", "payee_name", "apply_deadline", "required_docs", "where_to_apply", "summary"]);
const REQUIRED: Record<string, string[]> = {
  pay: ["issuer", "amount_krw", "due_date"], apply: ["issuer", "apply_deadline"], info: ["issuer"],
};
const CORE = ["amount_krw", "due_date", "apply_deadline", "issuer"];

export const normPhone = (p: unknown) => String(p ?? "").replace(/[^0-9]/g, "");
export const isPersonalMobile = (p: unknown) => MOBILE.test(normPhone(p));

export function normHost(u: unknown): string {
  if (u === null || u === undefined) return "";
  const raw = String(u).trim();
  if (!raw) return "";
  if (raw.includes("\\") || CTRL.test(raw)) return "";
  let s = raw;
  if (s.includes("://")) {
    if (!ALLOWED_SCHEME.has(s.split("://")[0].toLowerCase())) return "";
  } else {
    s = "http://" + s;
  }
  try {
    const url = new URL(s);
    if (url.username || url.password) return "";
    if (url.port) {
      const p = Number(url.port);
      if (!Number.isInteger(p) || p < 1 || p > 65535) return "";
    }
    const h = url.hostname.toLowerCase();
    if (!h || h.includes(" ")) return "";
    return h.startsWith("www.") ? h.slice(4) : h;
  } catch {
    return "";
  }
}

export function findIssuer(name: unknown): Issuer | null {
  if (name === null || name === undefined) return null;
  const n = String(name).replace(/\s+/g, "");
  if (n.length < MIN_ALIAS) return null;
  let best: [Issuer, number] | null = null;
  for (const it of REGISTRY.issuers) {
    for (const a of it.aliases) {
      const na = a.replace(/\s+/g, "");
      if (na.length < MIN_ALIAS) continue;
      if (n === na) return it;
      if (n.startsWith(na) && BRANCH_TAIL.test(n.slice(na.length))) {
        if (!best || na.length > best[1]) best = [it, na.length];
      }
    }
  }
  return best ? best[0] : null;
}

type Step = { value: unknown; av: Record<string, any> };

function pickSteps(job: any): Record<string, Step> {
  const steps: Record<string, Step> = {};
  for (const o of job?.output ?? []) {
    const m = String(o?.model ?? "").toLowerCase();
    const c = o?.content?.[0] ?? {};
    let av = c.additional_values ?? {};
    if (typeof av === "string") { try { av = JSON.parse(av); } catch { av = {}; } }
    const txt = c.text;
    let value: unknown = txt;
    if (typeof txt === "string" && txt.trim().startsWith("{")) { try { value = JSON.parse(txt); } catch { /* 원문 유지 */ } }
    const key = m.includes("parse") ? "parse"
      : m.includes("classify") ? "classify"
      : m.includes("extract") ? "extract"
      : m.includes("instruct") ? "instruct"
      : m.includes("validate") ? "validate" : null;
    if (key) steps[key] = { value, av };
  }
  return steps;
}

export function verify(job: any): VerifyResult {
  const base: VerifyResult = { verdict: "failed", checks: [], reasons: [] };
  if (job?.status !== "completed") return { ...base, reason: `job status=${job?.status}` };
  for (const o of job?.output ?? []) {
    if (o?.status && o.status !== "completed") return { ...base, reason: `step ${o.model} status=${o.status}` };
  }
  const st = pickSteps(job);
  if (!st.classify) return { ...base, reason: "classify 단계 없음" };

  const out: VerifyResult = { ...base };
  const docType = String(st.classify.value ?? "");
  const clsAv = st.classify.av?.document_type ?? {};
  out.actionType = docType;
  out.classifyConfidence = clsAv.confidence;
  out.classifyScore = clsAv.confidence_score;

  if (clsAv.confidence === "low") {
    out.verdict = "needs_human";
    out.reasons.push({ rule: "R5", detail: `문서 종류를 확신하지 못함 (분류 신뢰도 ${clsAv.confidence_score})`, action: "자녀가 직접 확인" });
    return out;
  }
  if (!st.extract) {
    if (["pay", "apply", "info"].includes(docType)) {
      return { ...out, verdict: "failed", reason: `'${docType}' 문서인데 Extract 단계가 없음 — 파이프라인 오류` };
    }
    out.verdict = "no_extract";
    return out;
  }

  const rawFields = st.extract.value;
  if (typeof rawFields !== "object" || rawFields === null || Array.isArray(rawFields)) {
    return { ...out, verdict: "failed", reason: "Extract 출력이 객체가 아님" };
  }
  const av = st.extract.av ?? {};
  const conf: Record<string, string> = {};
  for (const [k, v] of Object.entries(av)) {
    if (v && typeof v === "object" && "_value" in (v as any)) conf[k] = (v as any).confidence;
  }
  const dropped = Object.keys(rawFields).filter((k) => !ALLOW_FIELDS.has(k)).sort();
  const fields: Record<string, unknown> = {};
  for (const [k, v] of Object.entries(rawFields)) if (ALLOW_FIELDS.has(k)) fields[k] = v;
  out.fields = fields;
  out.fieldConfidence = conf;
  if (dropped.length) out.droppedFields = dropped;

  const missing = (REQUIRED[docType] ?? []).filter((k) => !String(fields[k] ?? "").trim());
  const lowCore = CORE.filter((k) => conf[k] === "low");
  out.speechSuppressed = lowCore.length > 0 || missing.length > 0;
  if (lowCore.length) out.reasons.push({ rule: "R3", detail: `핵심 필드 신뢰도 낮음: ${lowCore.join(", ")}`, action: "숫자 낭독 억제 + 자녀 확인" });
  if (missing.length) out.reasons.push({ rule: "R3", detail: `필수 필드 결손: ${missing.join(", ")}`, action: "숫자 낭독 억제 + 자녀 확인" });

  const phone = fields.contact_phone as string | undefined;
  const mobileHit = !!phone && isPersonalMobile(phone);
  const mobileStrong = mobileHit && conf.contact_phone === "high";
  if (mobileHit) {
    out.reasons.push({ rule: "R4", detail: `고지서 상담 번호가 개인 휴대전화입니다: ${phone}`, action: "이 번호로 연락하지 말 것. 기관 공식 대표번호만 사용" });
  }

  const issuer = findIssuer(fields.issuer);
  if (!issuer) {
    out.verdict = mobileStrong ? "mismatch" : "unknown_issuer";
    if (!mobileStrong) out.reasons.push({ rule: "R1", detail: `레지스트리에 없는 기관: ${fields.issuer}`, action: "판단 불가 — 자녀 확인" });
    if (phone) {
      out.checks.push({ name: "상담 번호 형식", value: phone, ok: conf.contact_phone === "high" ? !mobileHit : null, expected: ["기관 대표번호"], kind: "mobile" });
    }
    const r0 = out.checks.filter((c) => c.ok !== null);
    out.checksTotal = r0.length;
    out.checksPassed = r0.filter((c) => c.ok).length;
    return out;
  }

  let urlUnparseable = false;

  if (phone && phone.trim()) {
    const np = normPhone(phone);
    const exact = issuer.official_phones.some((x) => normPhone(x) === np);
    const prefix = (issuer.phone_prefixes ?? []).some((x) => x && np.startsWith(normPhone(x)));
    let ok: boolean | null = exact || prefix;
    let note: string | undefined = exact ? undefined : prefix ? "같은 국번 대역 — 부서 직통번호로 인정" : undefined;
    if (ok && conf.contact_phone !== "high") {
      ok = null;
      note = "읽기가 불확실해 대조 결과를 인정하지 않음";
      out.reasons.push({ rule: "R3", detail: "문의전화 읽기가 불확실함", action: "자녀 확인" });
    }
    out.checks.push({ name: "문의전화", value: phone, ok, note, expected: issuer.official_phones, conf: conf.contact_phone, kind: "phone" });
  } else {
    out.checks.push({ name: "문의전화", value: null, ok: null, expected: null, note: "문서에서 읽지 못함" });
  }

  const url = fields.info_url as string | undefined;
  if (url && url.trim()) {
    const h = normHost(url);
    if (!h) {
      out.checks.push({ name: "안내 주소", value: String(url).slice(0, 60), ok: null, expected: null, note: "주소 형식을 해석할 수 없음", kind: "host" });
      out.reasons.push({ rule: "R1", detail: `안내 주소를 해석할 수 없음: ${String(url).slice(0, 60)}`, action: "자녀 확인 — 문서의 링크를 열지 말 것" });
      urlUnparseable = true;
    } else {
      const exp = Array.from(new Set(issuer.official_hosts.map(normHost))).sort();
      let ok: boolean | null = exp.includes(h);
      if (ok && conf.info_url !== "high") ok = null;
      out.checks.push({ name: "안내 주소", value: h, ok, expected: exp, conf: conf.info_url, kind: "host" });
    }
  } else {
    out.checks.push({ name: "안내 주소", value: null, ok: null, expected: null, note: "문서에 없음" });
  }

  if (phone && phone.trim()) {
    out.checks.push({
      name: "상담 번호 형식", value: phone,
      ok: conf.contact_phone === "high" ? !mobileHit : null,
      note: conf.contact_phone === "high" ? undefined : "읽기가 불확실함",
      expected: ["기관 대표번호"], kind: "mobile",
    });
  }

  const payee = fields.payee_name as string | undefined;
  if (payee && payee.trim()) {
    const pn = payee.replace(/[\s()[\]]|주식회사|㈜/g, "");
    const matched = findIssuer(payee);
    const same = (!!matched && matched.issuer_id === issuer.issuer_id)
      || issuer.aliases.some((a) => pn.includes(a.replace(/\s/g, "")));
    out.checks.push({ name: "가상계좌 예금주", value: payee, ok: same, expected: [issuer.display_name], conf: conf.payee_name, kind: "payee" });
    if (!same) {
      out.reasons.push({ rule: "R6", detail: `가상계좌 예금주가 발급기관과 다릅니다: '${payee}' (발급기관 ${issuer.display_name})`, action: "이 계좌로 송금하지 말 것. 기관 공식 대표번호로 사실 확인" });
    }
  }

  out.checks.push({ name: "계좌 진위", value: null, ok: null, expected: null, note: "확인하지 않음 — 계좌 명의 조회 권한 없음" });

  const real = out.checks.filter((c) => c.ok !== null);
  out.checksTotal = real.length;
  out.checksPassed = real.filter((c) => c.ok).length;
  const failed = real.filter((c) => !c.ok);
  const hardFail = failed.filter((c) => c.conf !== "low" && ["mobile", "host", "payee"].includes(c.kind ?? ""));
  const softFail = failed.filter((c) => !hardFail.includes(c));

  if (!real.length) {
    out.verdict = "not_checkable";
    out.reasons.push({ rule: "R1", detail: "대조할 연락처·주소를 읽지 못함", action: "자녀 확인" });
  } else if (hardFail.length) {
    out.verdict = "mismatch";
    for (const c of hardFail) {
      out.reasons.push({ rule: "R1", detail: `${c.name} 불일치: 문서 '${c.value}' vs 공식 ${JSON.stringify(c.expected)}`, action: "공식 번호로만 연락. 문서의 번호·링크 사용 금지" });
    }
  } else if (softFail.length) {
    out.verdict = "review";
    for (const c of softFail) {
      const why = c.conf === "low" ? "읽기가 불확실함" : "등록된 공식 번호와 다름(부서 직통일 수 있음)";
      out.reasons.push({ rule: "R1", detail: `${c.name} ${why}: '${c.value}'`, action: "자녀 확인 — 기관 공식 대표번호로 사실 확인 권장" });
    }
  } else if (lowCore.length || missing.length || urlUnparseable) {
    out.verdict = "review";
  } else {
    out.verdict = "clear";
  }

  out.safeContact = { phones: issuer.official_phones, hosts: issuer.official_hosts, source: issuer.source_urls, verifiedAt: issuer.verified_at };
  return out;
}
```

```ts
const BASE = "https://api.upstage.ai/v2";
const TIMEOUT_MS = 20000;

function key(): string {
  const k = process.env.UPSTAGE_API_KEY;
  if (!k) throw new Error("UPSTAGE_API_KEY 없음");
  return k;
}

async function req(path: string, init: RequestInit): Promise<any> {
  const ctrl = new AbortController();
  const t = setTimeout(() => ctrl.abort(), TIMEOUT_MS);
  try {
    const res = await fetch(BASE + path, {
      ...init,
      signal: ctrl.signal,
      headers: { Authorization: `Bearer ${key()}`, ...(init.headers ?? {}) },
    });
    if (!res.ok) throw new Error(`Upstage ${res.status}: ${(await res.text()).slice(0, 300)}`);
    return res.json();
  } finally {
    clearTimeout(t);
  }
}

export async function uploadFile(buf: Buffer, filename: string, mime: string): Promise<string> {
  const fd = new FormData();
  fd.append("purpose", "user_data");
  fd.append("file", new Blob([buf], { type: mime }), filename);
  const j = await req("/files", { method: "POST", body: fd });
  return j.id;
}

export async function createJob(fileId: string) {
  return req("/responses", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: process.env.UPSTAGE_AGENT_ID,
      include: ["all"],
      input: [{ role: "user", content: [{ type: "input_file", file_id: fileId }] }],
    }),
  });
}

export async function fetchJob(jobId: string) {
  return req(`/responses/${encodeURIComponent(jobId)}?include[]=all`, { method: "GET" });
}

export async function deleteFile(fileId: string): Promise<void> {
  try {
    await req(`/files/${encodeURIComponent(fileId)}`, { method: "DELETE" });
  } catch {
    /* 삭제 실패는 흐름을 막지 않는다 */
  }
}
```

```ts
import type { VerifyResult } from "./types";

const D = ["영", "일", "이", "삼", "사", "오", "육", "칠", "팔", "구"];
const SMALL = ["", "십", "백", "천"];
const BIG = ["", "만", "억", "조"];

export function koNumber(n: number): string {
  if (!Number.isFinite(n)) return "";
  if (n === 0) return "영";
  const groups: string[] = [];
  let rest = Math.floor(Math.abs(n));
  let gi = 0;
  while (rest > 0) {
    const g = rest % 10000;
    if (g > 0) {
      const ds = String(g).split("").map(Number);
      let s = "";
      ds.forEach((d, i) => {
        const pos = ds.length - 1 - i;
        if (d === 0) return;
        s += (d === 1 && pos > 0 ? "" : D[d]) + SMALL[pos];
      });
      groups.unshift(s + BIG[gi]);
    }
    rest = Math.floor(rest / 10000);
    gi++;
  }
  return groups.join(" ");
}

export const koMoney = (n: number) => `${koNumber(n)} 원`;

export function koDate(iso: string): string {
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(iso ?? "").trim());
  if (!m) return "";
  return `${koNumber(Number(m[2]))}월 ${koNumber(Number(m[3]))}일`;
}

export const koPhone = (p: string) =>
  String(p ?? "").split("-")
    .map((part) => part.split("").map((c) => (c === "0" ? "공" : D[Number(c)] ?? c)).join(""))
    .join(" ");

function money(n: number): string {
  if (n < 10000) return `${n.toLocaleString()}원`;
  const man = Math.floor(n / 10000);
  const rest = n % 10000;
  return rest ? `${man}만 ${rest.toLocaleString()}원` : `${man}만원`;
}

function shortDate(iso: string): string {
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(iso ?? ""));
  return m ? `${Number(m[2])}월 ${Number(m[3])}일` : "";
}

const LABELS: [RegExp, string][] = [
  [/건강보험/, "건강보험료"], [/장기요양/, "장기요양보험료"], [/전기/, "전기요금"],
  [/가스/, "가스요금"], [/주민세/, "주민세"], [/지방세|체납/, "지방세"],
  [/통신|이동전화/, "통신요금"], [/기초연금/, "기초연금"], [/에너지바우처/, "에너지바우처"],
];

export function shortLabel(title: unknown): string {
  const t = String(title ?? "");
  for (const [re, label] of LABELS) if (re.test(t)) return label;
  return t.slice(0, 10) || "우편물";
}

export function buildPhrases(r: VerifyResult) {
  const f = r.fields ?? {};
  const docLabel = shortLabel(f.doc_title);
  const amount = typeof f.amount_krw === "number" ? f.amount_krw : null;
  const due = String(f.due_date ?? f.apply_deadline ?? "");

  if (r.verdict === "mismatch") {
    return {
      docLabel,
      screenLines: ["이 문서의 정보가", "공식 정보와 다릅니다"],
      speech: "어르신, 잠깐만요. 이 종이에 적힌 연락처가 공식 정보와 달라요. 여기 적힌 번호로는 전화하지 마세요. 자녀분께 먼저 여쭤보세요.",
    };
  }
  if (r.verdict === "needs_human" || r.verdict === "unknown_issuer") {
    return {
      docLabel,
      screenLines: ["자녀분께 확인을", "부탁드렸어요"],
      speech: "어르신, 이 문서는 제가 확실하게 읽지 못했어요. 자녀분께 확인을 부탁드렸어요.",
    };
  }
  if (r.speechSuppressed) {
    return {
      docLabel,
      screenLines: ["정확히 읽지 못했어요", "자녀분께 확인을 부탁드렸어요"],
      speech: `어르신, ${docLabel} 관련 우편물이에요. 내용을 정확히 읽지 못해서 자녀분께 확인을 부탁드렸어요.`,
    };
  }

  const screenLines: string[] = [];
  const spoken: string[] = [`어르신, ${docLabel} 우편물이에요.`];
  if (due) screenLines.push(`${shortDate(due)}까지`);
  if (amount !== null) screenLines.push(money(amount));
  if (amount !== null && due) spoken.push(`문서에 적힌 금액은 ${koMoney(amount)}이고, 기한은 ${koDate(due)}입니다.`);
  else if (due) spoken.push(`기한은 ${koDate(due)}입니다.`);
  spoken.push("자녀분께 보내드렸어요.");
  return { docLabel, screenLines, speech: spoken.join(" ") };
}
```

```ts
export async function POST(req: Request) {
  const form = await req.formData();
  const file = form.get("file");
  if (!(file instanceof File)) return NextResponse.json({ error: "file 없음" }, { status: 400 });
  if (file.size > 8 * 1024 * 1024) return NextResponse.json({ error: "파일이 너무 큽니다" }, { status: 413 });

      .insert({ household_id: HOUSEHOLD, upstage_job_id: job.id, upstage_file_id: fileId, pipeline_status: job.status })

    return NextResponse.json({ id: data.id, jobId: job.id, status: "processing" }, { status: 202 });
```

```ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { fetchJob, deleteFile } from "@/lib/upstage";
import { verify } from "@/lib/verify";
import { buildPhrases } from "@/lib/phrase";

export const runtime = "nodejs";

export async function GET(_req: Request, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;
  const { data: doc } = await db.from("documents").select("*").eq("id", id).single();
  if (!doc) return NextResponse.json({ error: "없음" }, { status: 404 });
  if (doc.result) return NextResponse.json(doc);

  let job: any;
  try {
    job = await fetchJob(doc.upstage_job_id); // 1회만. 폴링은 클라이언트가 한다
  } catch (e) {
    return NextResponse.json({ ...doc, pipeline_status: "error", error: String(e).slice(0, 200) }, { status: 200 });
  }

  if (job.status === "queued" || job.status === "in_progress") {
    await db.from("documents").update({ pipeline_status: job.status }).eq("id", id);
    return NextResponse.json({ ...doc, pipeline_status: job.status });
  }

  const result = verify(job);
  const phrases = buildPhrases(result);
  const { data: updated } = await db.from("documents")
    .update({ pipeline_status: job.status, action_type: result.actionType, verdict: result.verdict, result, phrases })
    .eq("id", id).select("*").single();
  if (doc.upstage_file_id) await deleteFile(doc.upstage_file_id); // 원본은 남기지 않는다
  return NextResponse.json(updated);
}

export async function PATCH(req: Request, { params }: { params: Promise<{ id: string }> }) {
  const { id } = await params;
  const { resolution } = await req.json();
  if (!["acknowledged", "done"].includes(resolution)) {
    return NextResponse.json({ error: "잘못된 값" }, { status: 400 });
  }
  const patch: Record<string, unknown> = { resolution_status: resolution };
  patch[resolution === "done" ? "done_at" : "reviewed_at"] = new Date().toISOString();
  const { data } = await db.from("documents").update(patch).eq("id", id).select("*").single();
  return NextResponse.json(data);
}
```

```ts
export async function compress(file: File, maxDim = 1600, quality = 0.82): Promise<File> {
  const bmp = await createImageBitmap(file);
  const scale = Math.min(1, maxDim / Math.max(bmp.width, bmp.height));
  const w = Math.round(bmp.width * scale);
  const h = Math.round(bmp.height * scale);
  const canvas = document.createElement("canvas");
  canvas.width = w;
  canvas.height = h;
  canvas.getContext("2d")!.drawImage(bmp, 0, 0, w, h);
  const blob: Blob = await new Promise((res) => canvas.toBlob((b) => res(b!), "image/jpeg", quality));
  return new File([blob], "mail.jpg", { type: "image/jpeg" });
}
```

`lib/welfare.ts`:

```ts
import snapshot from "@/data/welfare_snapshot.json";
import { findIssuer } from "./verify";
import type { VerifyResult } from "./types";

export type Benefit = {
  id: string; name: string; agency: string; target: string;
  howTo: string; contact: string; sourceUrl: string; verifiedAt: string;
};
type Row = Benefit & { triggers: string[] };

const ROWS = (snapshot as unknown as { benefits: Row[] }).benefits;
export const WELFARE_COVERAGE = (snapshot as unknown as { coverage_note: string }).coverage_note;

export function findRelatedBenefits(result: VerifyResult): Benefit[] {
  if (!result.fields) return [];
  if (["ad", "other"].includes(result.actionType ?? "")) return [];
  const issuer = findIssuer(result.fields.issuer);
  if (!issuer) return [];
  return ROWS.filter((r) => r.triggers.includes(issuer.issuer_id))
    .map(({ triggers: _t, ...b }) => b);
}
```
